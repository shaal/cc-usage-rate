<project_specification>
  <project_name>Claude Usage Tracker Chrome Extension</project_name>

  <overview>
    A Chrome extension that enhances the Claude.ai usage settings page by providing visual feedback on usage efficiency. The extension analyzes the user&apos;s current session usage and weekly usage limits, comparing actual usage percentages against expected time-based percentages. It displays color-coded indicators (green/yellow/red) with circular badges to show whether the user is under-utilizing, optimally utilizing, or over-utilizing their Claude quota relative to the time elapsed in the current usage window. This helps users pace their Claude usage to avoid hitting limits prematurely or leaving unused capacity on the table.
  </overview>

  <technology_stack>
    <technology>Chrome Extension Manifest V3</technology>
    <technology>JavaScript/TypeScript</technology>
    <technology>Chrome Content Scripts API</technology>
    <technology>Chrome Storage API</technology>
    <technology>HTML5 Canvas or SVG for circular indicators</technology>
    <technology>CSS3 for styling and animations</technology>
    <technology>Webpack or Vite for bundling</technology>
  </technology_stack>

  <core_capabilities>
    <capability>Detect and parse usage statistics from claude.ai/settings/usage page</capability>
    <capability>Calculate session usage efficiency by comparing usage percentage vs time elapsed in 5-hour window</capability>
    <capability>Calculate weekly usage efficiency by comparing usage percentage vs time elapsed in weekly window</capability>
    <capability>Display color-coded circular indicators (green/yellow/red) showing relative efficiency</capability>
    <capability>Automatically update calculations when page content changes</capability>
    <capability>Parse time remaining strings to calculate elapsed time percentages</capability>
    <capability>Non-intrusive UI injection that matches Claude&apos;s design aesthetic</capability>
  </core_capabilities>

  <implemented_features>
    <feature>
      <name>Session Usage Tracker</name>
      <description>Monitors the &apos;Current session&apos; usage section, extracts the reset timer (e.g., &apos;2 hr 48 min&apos;) and usage percentage (e.g., &apos;24%&apos;). Calculates the time elapsed in the 5-hour window, determines the expected usage percentage based on elapsed time, and displays a circular indicator showing the difference. Green indicates usage is below expected (pacing well), yellow indicates usage is near expected (neutral), red indicates usage exceeds expected (at risk of hitting limits).</description>
      <file_locations>
        <location>src/content/sessionTracker.ts</location>
        <location>src/utils/timeCalculations.ts</location>
      </file_locations>
    </feature>
    <feature>
      <name>Weekly Usage Tracker</name>
      <description>Monitors the &apos;Weekly limits&apos; usage section, extracts the reset day/time (e.g., &apos;Thu 7:59 AM&apos;) and usage percentage (e.g., &apos;45%&apos;). Calculates how much of the week has elapsed since the last reset, determines expected usage percentage, and displays a color-coded circular indicator showing the relative efficiency for weekly pacing.</description>
      <file_locations>
        <location>src/content/weeklyTracker.ts</location>
        <location>src/utils/weeklyCalculations.ts</location>
      </file_locations>
    </feature>
    <feature>
      <name>Color-Coded Circular Indicators</name>
      <description>Renders visual circular badges next to usage statistics. The indicator displays a percentage number inside a colored circle. Color coding: Green (usage is 10%+ below time-based expectation - room to use more), Yellow (usage is within ±10% of expectation - on track), Red (usage is 10%+ above expectation - over-consuming relative to time). The number displayed is the difference percentage (e.g., &apos;+15%&apos; or &apos;-20%&apos;).</description>
      <file_locations>
        <location>src/components/UsageIndicator.ts</location>
        <location>src/styles/indicator.css</location>
      </file_locations>
    </feature>
    <feature>
      <name>DOM Observer and Auto-Update</name>
      <description>Uses MutationObserver to detect changes in the usage page content, automatically recalculating and updating indicators when the timer counts down or usage values change without requiring page refresh.</description>
      <file_locations>
        <location>src/content/observer.ts</location>
      </file_locations>
    </feature>
    <feature>
      <name>Time Parsing Utilities</name>
      <description>Parses various time formats from the page including relative times (&apos;2 hr 48 min&apos;), day-based resets (&apos;Thu 7:59 AM&apos;), and percentage strings. Converts these to calculable values for efficiency comparisons.</description>
      <file_locations>
        <location>src/utils/timeParser.ts</location>
      </file_locations>
    </feature>
  </implemented_features>
